rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAuth() {
      return request.auth != null;
    }

    match /users/{userId} {
      allow read: if isAuth();
      allow create, update: if isAuth() && request.auth.uid == userId;
      allow delete: if false;
    }

    match /lists/{listId} {
      allow read: if isAuth();
      allow create: if isAuth() && request.auth.uid in request.resource.data.memberIds;
      allow update: if isAuth() && (request.auth.uid in resource.data.memberIds || resource.data.owner == request.auth.uid);
      allow delete: if isAuth() && resource.data.owner == request.auth.uid;
    }

    match /tasks/{taskId} {
      allow read, create, update, delete: if isAuth();
    }

    match /invitations/{invitationId} {
      allow read: if true;
      allow create, update, delete: if isAuth();
    }

    match /notifications/{notificationId} {
      allow read, update, delete: if isAuth() && resource.data.userId == request.auth.uid;
      allow create: if isAuth();
    }

  }
}
